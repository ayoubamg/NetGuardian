! SW1 - Layer 2 access switch
! Adjust interface names and port assignments to match your switch model.
enable
configure terminal
hostname SW1
no ip domain-lookup

vlan 10
 name ADMIN
vlan 20
 name IT
vlan 30
 name EMPLOYEES
vlan 40
 name GUESTS
vlan 99
 name MANAGEMENT
!
interface GigabitEthernet0/1
 description TRUNK_TO_R1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,99
 no shutdown
!
interface FastEthernet0/1
 description ADMIN_PC
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface FastEthernet0/2
 description IT_PC
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
!
interface FastEthernet0/3
 description EMPLOYEE_PC
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
!
interface FastEthernet0/4
 description GUEST_PC
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
!
interface Vlan99
 description SWITCH_MANAGEMENT
 ip address 192.168.99.2 255.255.255.0
 no shutdown
!
ip default-gateway 192.168.99.1
end
write memory

! Validation commands:
! show vlan brief
! show interfaces trunk
! show ip interface brief
