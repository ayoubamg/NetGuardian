! R1 - Enterprise Network Lab
! Adjust interface names to match your Packet Tracer router model.
enable
configure terminal
hostname R1
no ip domain-lookup

interface GigabitEthernet0/0
 no shutdown
 description TRUNK_TO_SW1
!
interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
!
interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
!
interface GigabitEthernet0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
!
interface GigabitEthernet0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip access-group GUEST-IN in
!
interface GigabitEthernet0/0.99
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
!
ip dhcp excluded-address 192.168.10.1 192.168.10.20
ip dhcp excluded-address 192.168.20.1 192.168.20.20
ip dhcp excluded-address 192.168.30.1 192.168.30.20
ip dhcp excluded-address 192.168.40.1 192.168.40.20
ip dhcp excluded-address 192.168.99.1 192.168.99.20
!
ip dhcp pool ADMIN
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
!
ip dhcp pool IT
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8
!
ip dhcp pool EMPLOYEES
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 8.8.8.8
!
ip dhcp pool GUESTS
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 8.8.8.8
!
ip dhcp pool MANAGEMENT
 network 192.168.99.0 255.255.255.0
 default-router 192.168.99.1
 dns-server 8.8.8.8
!
ip access-list extended GUEST-IN
 remark Block Guest VLAN from reaching internal private networks
 deny ip 192.168.40.0 0.0.0.255 10.0.0.0 0.255.255.255
 deny ip 192.168.40.0 0.0.0.255 172.16.0.0 0.15.255.255
 deny ip 192.168.40.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip 192.168.40.0 0.0.0.255 any
!
end
write memory

! Validation commands:
! show ip interface brief
! show ip route
! show ip dhcp binding
! show access-lists
